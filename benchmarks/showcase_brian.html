// src/kernel/intern.rs — The NUMBERS File
use std::collections::HashMap;

#[derive(Default, Debug, Clone)]
pub struct StringInterner {
    // Map string to dense, sequential u32 identifier
    map: HashMap<String, u32>,
    names: Vec<String>,
}

impl StringInterner {
    /// Allocation occurs once upon insertion; returns raw 32-bit integer
    pub fn intern(&mut self, s: &str) -> u32 {
        if let Some(&id) = self.map.get(s) {
            return id;
        }
        let id = self.names.len() as u32;
        self.map.insert(s.to_string(), id);
        self.names.push(s.to_string());
        id
    }

    #[inline(always)]
    pub fn get(&self, s: &str) -> Option<u32> {
        self.map.get(s).copied()
    }
}

// src/kernel/bitmask.rs — The BITS File
pub type QueryMask = u64;

/// Calculate uncovered query terms provided by a candidate turn.
/// Executes in 1 CPU clock cycle using native x86 POPCNT instruction.
#[inline(always)]
pub fn uncovered_query_term_count(
    turn_mask: QueryMask, 
    covered_mask: QueryMask, 
    total_groups: usize
) -> u32 {
    // Fast bitwise AND-NOT: finds query bits present in turn but not yet covered
    let uncovered_bits = turn_mask & !covered_mask;
    
    let group_boundary_mask = if total_groups >= 64 {
        !0u64
    } else {
        (1u64 << total_groups) - 1
    };
    
    // Lowers directly to hardware POPCNT
    (uncovered_bits & group_boundary_mask).count_ones()
}

/// Marginal Information Gain ratio (0.0 to 1.0)
#[inline(always)]
pub fn marginal_information_ratio(
    turn_mask: QueryMask, 
    covered_mask: QueryMask, 
    total_groups: usize
) -> f32 {
    if total_groups == 0 { return 0.0; }
    let count = uncovered_query_term_count(turn_mask, covered_mask, total_groups);
    count as f32 / total_groups as f32
}
